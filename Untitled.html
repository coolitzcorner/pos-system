  }


  function removeFromCart(id) {

    const index = cart.findIndex(c => c.id === id);

    if (index !== -1) {

      const removed = cart[index];

      cart.splice(index, 1);

      updateCartUI();

      showToast(`🗑️ ${removed.name} removed`, 'info');

    }

  }


  function changeQty(id, delta) {

    const item = cart.find(c => c.id === id);

    if (!item) return;

    const stockItem = inventoryData.find(i => i.id === id);

    const newQty = item.qty + delta;

    if (newQty <= 0) { removeFromCart(id); return; }

    if (newQty > stockItem.stock) {

      showToast(`⚠️ Not enough stock. Only ${stockItem.stock} left.`, 'warning');

      return;

    }

    item.qty = newQty;

    updateCartUI();

  }


  function clearCart() {

    if (cart.length === 0) { showToast('Cart is already empty', 'info'); return; }

    cart = [];

    updateCartUI();

    showToast('🧹 Cart cleared', 'info');

  }


  function getTotal() { return cart.reduce((sum, item) => sum + (item.price * item.qty), 0); }

  function getTotalItems() { return cart.reduce((sum, item) => sum + item.qty, 0); }


  function updateCartUI() {

    if (cart.length === 0) {

      cartList.innerHTML = `<div class="empty-cart-msg"><i class="fas fa-egg" style="font-size: 2rem; display: block; margin-bottom: 10px;"></i> Your cart is empty<br><span style="font-size:0.8rem;">Add items from the menu</span></div>`;

    } else {

      cartList.innerHTML = cart.map(item => `

        <div class="cart-item" data-id="${item.id}">

          <div class="cart-item-left">

            <span class="cart-item-name">${item.emoji} ${item.name}</span>

            <div class="cart-item-meta"><span>₱${item.price}</span><span>·</span><span>${item.category}</span></div>

          </div>

          <div class="cart-item-actions">

            <button class="qty-btn" data-action="decr" data-id="${item.id}">−</button>

            <span class="qty-num">${item.qty}</span>

            <button class="qty-btn" data-action="incr" data-id="${item.id}">+</button>

            <button class="remove-item" data-action="remove" data-id="${item.id}"><i class="fas fa-times"></i></button>

          </div>

        </div>

      `).join('');


      document.querySelectorAll('.cart-item').forEach(el => {

        const id = parseInt(el.dataset.id);

        el.querySelector('[data-action="incr"]')?.addEventListener('click', (e) => { e.stopPropagation(); changeQty(id, 1); });

        el.querySelector('[data-action="decr"]')?.addEventListener('click', (e) => { e.stopPropagation(); changeQty(id, -1); });

        el.querySelector('[data-action="remove"]')?.addEventListener('click', (e) => { e.stopPropagation(); removeFromCart(id); });

      });

    }


    const total = getTotal();

    const totalItems = getTotalItems();

    cartTotalAmount.textContent = `₱${total.toFixed(2)}`;

    cartItemCount.textContent = totalItems;

    cartCountHeader.textContent = totalItems;

    totalHeader.textContent = `₱${total.toFixed(2)}`;

  }
